     │                                                                         │
     │ import { Middleware } from '@reduxjs/toolkit';                          │
     │                                                                         │
     │ // Track pending debounced actions                                      │
     │ const debouncedActions: Record<string, NodeJS.Timeout> = {};            │
     │                                                                         │
     │ export const debounceMiddleware: Middleware = (store) => (next) =>      │
     │ (action) => {                                                           │
     │   // Check if action has debounce metadata                              │
     │   if (action.meta?.debounce) {                                          │
     │     const { delay, key } = action.meta.debounce;                        │
     │                                                                         │
     │     // Clear existing timeout for this key                              │
     │     if (debouncedActions[key]) {                                        │
     │       clearTimeout(debouncedActions[key]);                              │
     │     }                                                                   │
     │                                                                         │
     │     // Set new timeout                                                  │
     │     debouncedActions[key] = setTimeout(() => {                          │
     │       // Remove debounce metadata before dispatching                    │
     │       const { meta, ...actionWithoutMeta } = action;                    │
     │       next(actionWithoutMeta);                                          │
     │       delete debouncedActions[key];                                     │
     │     }, delay);                                                          │
     │                                                                         │
     │     return; // Don't dispatch immediately                               │
     │   }                                                                     │
     │                                                                         │
     │   return next(action);                                                  │
     │ };                                                                      │
     │                                                                         │
     │ Usage in slices:                                                        │
     │ // In component:                                                        │
     │ dispatch(autoSave({                                                     │
     │   meta: { debounce: { delay: 2000, key: 'autoSave' } }                  │
     │ }));                                                                    │

     │ frontend/lib/redux/middleware/cacheMiddleware.ts                        │
     │                                                                         │
     │ import { Middleware } from '@reduxjs/toolkit';                          │
     │                                                                         │
     │ interface CacheEntry {                                                  │
     │   data: any;                                                            │
     │   timestamp: number;                                                    │
     │   ttl: number;                                                          │
     │ }                                                                       │
     │                                                                         │
     │ const cache = new Map<string, CacheEntry>();                            │
     │                                                                         │
     │ export const cacheMiddleware: Middleware = (store) => (next) => (action)│
     │  => {                                                                   │
     │   // Check if action requests cached data                               │
     │   if (action.meta?.cache) {                                             │
     │     const { key, ttl = 300000 } = action.meta.cache; // Default 5min TTL│
     │                                                                         │
     │     const cached = cache.get(key);                                      │
     │     if (cached && Date.now() - cached.timestamp < cached.ttl) {         │
     │       // Return cached data without dispatching action                  │
     │       return Promise.resolve(cached.data);                              │
     │     }                                                                   │
     │   }                                                                     │
     │                                                                         │
     │   const result = next(action);                                          │
     │                                                                         │
     │   // Cache successful async action results                              │
     │   if (action.meta?.cache && result instanceof Promise) {                │
     │     result.then((data) => {                                             │
     │       const { key, ttl = 300000 } = action.meta.cache;                  │
     │       cache.set(key, {                                                  │
     │         data,                                                           │
     │         timestamp: Date.now(),                                          │
     │         ttl,                                                            │
     │       });                                                               │
     │     });                                                                 │
     │   }                                                                     │
     │                                                                         │
     │   return result;                                                        │
     │ };                                                                      │
     │                                                                         │
     │ // Cache invalidation                                                   │
     │ export const invalidateCache = (key: string) => {                       │
     │   cache.delete(key);                                                    │
     │ };                                                                      │
     │                                                                         │
     │ export const invalidateCachePattern = (pattern: string) => {            │
     │   for (const key of cache.keys()) {                                     │
     │     if (key.includes(pattern)) {                                        │
     │       cache.delete(key);                                                │
     │     }                                                                   │
     │   }                                                                     │
     │ };                                                                      │

     │ export const calculateFBLR = (                                          │
     │   position: SpreadsheetPosition,                                        │
     │   rates: IndirectRates,                                                 │
     │   escalationRates: EscalationRates,                                     │
     │   year: string                                                          │
     │ ): number => {                                                          │
     │   // Pure calculation logic                                             │
     │   const wage = position.custom_salary ||                                │
     │ position[`wage_${position.percentile}`] || position.selected_wage || 0; │
     │   const hours = position.hours_per_year[year] || 0;                     │
     │                                                                         │
     │   if (hours === 0) return 0;                                            │
     │                                                                         │
     │   const dlRate = wage / hours;                                          │
     │   const fringe = dlRate * rates.fringe;                                 │
     │   const oh = (dlRate + fringe) * rates.oh;                              │
     │   const ga = (dlRate + fringe + oh) * rates.ga;                         │
     │   const fee = (dlRate + fringe + oh + ga) * rates.fee;                  │
     │                                                                         │
     │   return dlRate + fringe + oh + ga + fee;                               │
     │ };                                                                      │

     │ export const transformPositionsToAdvanced = (                           │
     │   positions: SpreadsheetPosition[],                                     │
     │   rates: IndirectRates,                                                 │
     │   escalationRates: EscalationRates                                      │
     │ ): AdvancedPosition[] => {                                              │
     │   return positions.map((pos) => {                                       │
     │     const breakdown: AdvancedPosition['breakdown'] = {};                │
     │                                                                         │
     │     Object.entries(pos.hours_per_year).forEach(([year, hours]) => {     │
     │       const wage = pos.custom_salary || pos[`wage_${pos.percentile}`] ||│
     │  pos.selected_wage || 0;                                                │
     │       const dlRate = hours > 0 ? wage / hours : 0;                      │
     │       const dlAmount = dlRate * hours;                                  │
     │                                                                         │
     │       const fringe = dlRate * rates.fringe;                             │
     │       const fringeAmount = fringe * hours;                              │
     │                                                                         │
     │       const oh = (dlRate + fringe) * rates.oh;                          │
     │       const ohAmount = oh * hours;                                      │
     │                                                                         │
     │       const ga = (dlRate + fringe + oh) * rates.ga;                     │
     │       const gaAmount = ga * hours;                                      │
     │                                                                         │
     │       const fee = (dlRate + fringe + oh + ga) * rates.fee;              │
     │       const feeAmount = fee * hours;                                    │
     │                                                                         │
     │       const fblr = dlRate + fringe + oh + ga + fee;                     │
     │       const totalAmount = fblr * hours;                                 │
     │                                                                         │
     │       breakdown[year] = {                                               │
     │         hours,                                                          │
     │         wage,                                                           │
     │         dlRate,                                                         │
     │         dlAmount,                                                       │
     │         fringe,                                                         │
     │         fringeAmount,                                                   │
     │         oh,                                                             │
     │         ohAmount,                                                       │
     │         ga,                                                             │
     │         gaAmount,                                                       │
     │         fee,                                                            │
     │         feeAmount,                                                      │
     │         fblr,                                                           │
     │         totalAmount,                                                    │
     │       };                                                                │
     │     });                                                                 │
     │                                                                         │
     │     return {                                                            │
     │       ...pos,                                                           │
     │       breakdown,                                                        │
     │       total_hours: Object.values(pos.hours_per_year).reduce((sum, h) => │
     │ sum + h, 0),                                                            │
     │       total_amount: Object.values(breakdown).reduce((sum, b) => sum +   │
     │ b.totalAmount, 0),                                                      │
     │     } as AdvancedPosition;                                              │
     │   });                                                                   │
     │ };                                                                      │

     │ export const calculateAggregates = (positionsAdvanced:                  │
     │ AdvancedPosition[]): Aggregates => {                                    │
     │   const aggregates: Aggregates = {                                      │
     │     totalDL: 0,                                                         │
     │     totalFringe: 0,                                                     │
     │     totalOH: 0,                                                         │
     │     totalGA: 0,                                                         │
     │     totalFBLR: 0,                                                       │
     │     byYear: {},                                                         │
     │   };                                                                    │
     │                                                                         │
     │   positionsAdvanced.forEach((pos) => {                                  │
     │     Object.entries(pos.breakdown).forEach(([year, breakdown]) => {      │
     │       if (!aggregates.byYear[year]) {                                   │
     │         aggregates.byYear[year] = {                                     │
     │           dl: 0,                                                        │
     │           fringe: 0,                                                    │
     │           oh: 0,                                                        │
     │           ga: 0,                                                        │
     │           fblr: 0,                                                      │
     │           totalAmount: 0,                                               │
     │         };                                                              │
     │       }                                                                 │
     │                                                                         │
     │       aggregates.byYear[year].dl += breakdown.dlAmount;                 │
     │       aggregates.byYear[year].fringe += breakdown.fringeAmount;         │
     │       aggregates.byYear[year].oh += breakdown.ohAmount;                 │
     │       aggregates.byYear[year].ga += breakdown.gaAmount;                 │
     │       aggregates.byYear[year].fblr += breakdown.totalAmount;            │

     │       aggregates.totalDL += breakdown.dlAmount;                         │
     │       aggregates.totalFringe += breakdown.fringeAmount;                 │
     │       aggregates.totalOH += breakdown.ohAmount;                         │
     │       aggregates.totalGA += breakdown.gaAmount;                         │
     │       aggregates.totalFBLR += breakdown.totalAmount;                    │
