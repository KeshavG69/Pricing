  Afternoon: The Blocking Modal

  This is the most important component! You create frontend/components/terms/TermsBlockingModal.tsx:

  'use client';

  import { useState, useEffect } from 'react';
  import { termsApi } from '@/lib/api/terms';
  import { useAuthStore } from '@/lib/stores/authStore';
  import ReactMarkdown from 'react-markdown';
  import Button from '@/components/ui/Button';

  export function TermsBlockingModal() {
    const { user, fetchUser } = useAuthStore();
    const [termsContent, setTermsContent] = useState('');
    const [isAccepting, setIsAccepting] = useState(false);

    // Fetch terms content when modal opens
    useEffect(() => {
      if (user?.needs_terms_acceptance) {
        termsApi.getContent('terms').then(setTermsContent);
      }
    }, [user?.needs_terms_acceptance]);

    const handleAccept = async () => {
      setIsAccepting(true);
      try {
        await termsApi.acceptTerms();
        await fetchUser();  // Refresh user - needs_terms_acceptance will now be false
      } catch (error) {
        alert('Failed to accept terms');
      } finally {
        setIsAccepting(false);
      }
    };

    // Don't show modal if user doesn't need to accept
    if (!user?.needs_terms_acceptance) {
      return null;
    }

    return (
      <div className="fixed inset-0 z-50 flex items-center justify-center bg-black/80">
        {/* Modal - cannot close! */}
        <div className="bg-white rounded-lg w-[90vw] h-[90vh] flex flex-col p-6">

          {/* Header */}
          <div className="mb-4">
            <h2 className="text-2xl font-bold">Terms & Conditions Updated</h2>
            <p className="text-gray-600">
              You must accept the updated terms to continue using PriceIQ.
            </p>
          </div>

          {/* Scrollable content */}
          <div className="flex-1 overflow-auto border rounded p-4 bg-gray-50">
            <ReactMarkdown>{termsContent}</ReactMarkdown>
          </div>

          {/* Accept button */}
          <div className="mt-4">
            <Button
              onClick={handleAccept}
              isLoading={isAccepting}
              fullWidth
              variant="primary"
            >
              I Accept the Terms and Conditions
            </Button>
          </div>
        </div>
      </div>
    );
  }

  What this does:
  1. Checks if user.needs_terms_acceptance is true
  2. If yes → Fetches terms content from API
  3. Shows full-screen modal (can't close, can't click outside)
  4. When user clicks "I Accept" → Calls API → Refreshes user → Modal disappears
